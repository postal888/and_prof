async function loadVocabularyFromServer() {
  try {
    var r = await fetch('/api/sync/vocabulary', { credentials: 'include' });
    if (r.status === 204) { words = DEFAULT_VOCAB_SEED.slice(); recomputeNextIdFromWords(); return; }
    if (!r.ok) throw new Error('HTTP ' + r.status);
    var data = await r.json();
    var w = data && data.json && data.json.words;
    if (Array.isArray(w) && w.length) {
      words = w;
      recomputeNextIdFromWords();
      return;
    }

    console.warn('loadVocabularyFromServer', e);
  }
  words = DEFAULT_VOCAB_SEED.slice();
