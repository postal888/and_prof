function __syncBuildPayload(serverKey) {
  if (serverKey === 'settings') {
    var obj = {};
    __SETTINGS_KEYS.forEach(function (k) { var v = localStorage.getItem(k); if (v !== null) obj[k] = v; });
    return obj;
  }
  var lsKey = __lsKeyForServer(serverKey);
  var raw = lsKey ? localStorage.getItem(lsKey) : null;

}

function __syncPut(serverKey, jsonVal) {

}

function __syncScheduleKey(serverKey) {
  if (__syncTimers[serverKey]) clearTimeout(__syncTimers[serverKey]);
  __syncTimers[serverKey] = setTimeout(function () {
    __syncTimers[serverKey] = null;
    var payload = __syncBuildPayload(serverKey);
    if (payload == null) return;
    __syncPut(serverKey, payload);
  }, 700);
}
